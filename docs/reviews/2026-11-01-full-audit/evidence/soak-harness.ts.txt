// Soak harness (audit only, lives in the throwaway clean copy): repeated map travel, measures retained heap and RSS.
import { heapStats } from "bun:jsc";
import { Application } from "./src/app/bootstrap/application.ts";
import { parseApplicationCommand } from "./src/app/bootstrap/options.ts";

const corpus = process.env["CORPUS"] ?? "<game-data>";
const [scenario = "q1", cyclesArg = "20", stepsArg = "100", visual = ""] = Bun.argv.slice(2);
const configs: Record<string, { args: string[]; maps: string[]; bots?: string[] }> = {
  q1: { args: ["--game", "q1-classic-id1", "--map", "e1m1", "--movement", "q1", "--character", "q1", "--mode", "singleplayer"], maps: ["e1m2", "e1m1"] },
  q2: { args: ["--game", "q2-classic-baseq2", "--map", "base1", "--movement", "q2", "--character", "q2", "--mode", "singleplayer"], maps: ["base2", "base1"] },
  q3: { args: ["--game", "q3-baseq3", "--map", "q3dm1", "--movement", "q3", "--character", "q3", "--mode", "deathmatch"], maps: ["q3dm2", "q3dm1"], bots: ["ranger", "sarge", "visor", "major"] },
};
const config = configs[scenario]!;
const launch = parseApplicationCommand(["--content-root", corpus, ...config.args, ...(visual === "cpu" ? ["--renderer", "cpu", "--hidden", "--width", "320", "--height", "240"] : ["--dedicated"])]);
if (launch.kind !== "run") throw new Error("launch");
const app = await Application.open(launch.options, { print: () => undefined });
const snapshot = () => { Bun.gc(true); const s = heapStats(); return { heapMB: +(s.heapSize / 1048576).toFixed(1), objects: s.objectCount, rssMB: +(process.memoryUsage().rss / 1048576).toFixed(0), extMB: +(s.extraMemorySize / 1048576).toFixed(1) }; };
const report = (label: string, extra: object = {}) => console.log(JSON.stringify({ label, ...snapshot(), ...extra }));
const topTypes = () => { const c = heapStats().objectTypeCounts as Record<string, number>; return c; };
try {
  for (const bot of config.bots ?? []) app.queueCommand("addbot", [bot, "3"], null);
  const stepMsAll: number[] = [];
  for (let i = 0; i < Number(stepsArg); i++) await app.step(50);
  report("warm");
  const base = topTypes();
  // steady state: same map, no travel
  for (let round = 0; round < 3; round++) { const t0 = performance.now(); for (let i = 0; i < Number(stepsArg) * 3; i++) await app.step(50); report(`steady-${round}`, { msPerStep: +((performance.now() - t0) / (Number(stepsArg) * 3)).toFixed(3) }); }
  for (let cycle = 0; cycle < Number(cyclesArg); cycle++) {
    const map = config.maps[cycle % config.maps.length]!;
    const t0 = performance.now();
    app.queueCommand("map", [map], null);
    for (let i = 0; i < Number(stepsArg); i++) { const a = performance.now(); await app.step(50); stepMsAll.push(performance.now() - a); }
    report(`cycle-${cycle}-${map}`, { ms: Math.round(performance.now() - t0), players: app.simulation.players().length });
  }
  const end = topTypes();
  const deltas = Object.entries(end).map(([k, v]) => [k, v - (base[k] ?? 0)] as const).filter(([, d]) => Math.abs(d) > 200).sort((a, b) => b[1] - a[1]).slice(0, 30);
  console.log("TYPE-DELTAS", JSON.stringify(deltas));
  stepMsAll.sort((a, b) => a - b);
  console.log("STEP", JSON.stringify({ median: stepMsAll[stepMsAll.length >> 1], p99: stepMsAll[Math.floor(stepMsAll.length * 0.99)], max: stepMsAll.at(-1) }));
} finally { await app.close(); }
process.exit(0);
